// TRANSIENT macOS acceptance fixture. Not a production module or portable CI
// input. Append to store/installation_key.rs only in an isolated worktree,
// opt in explicitly, run this exact test, then remove the append before final
// formatting/build/commit. Never use a real installation's account identifier.
#[cfg(all(test, target_os = "macos"))]
mod disposable_keychain_acceptance {
    use super::*;
    use security_framework::passwords::{delete_generic_password, get_generic_password};

    struct OwnedItem {
        account: String,
        active: bool,
    }

    impl Drop for OwnedItem {
        fn drop(&mut self) {
            if self.active {
                if let Err(error) = delete_generic_password(super::super::KEYCHAIN_SERVICE, &self.account) {
                    eprintln!("disposable audit item cleanup failed for {}: {error}", self.account);
                }
            }
        }
    }

    #[test]
    fn create_only_adapter_preserves_generic_lookup_and_duplicate_winner() -> Result<()> {
        assert!(matches!(
            std::env::var("NATIVE_AUDIT_DISPOSABLE_KEYCHAIN").as_deref(), Ok("1")
        ), "explicit disposable-item acceptance opt-in required");
        let account = format!("audit-360-{}", uuid::Uuid::new_v4());
        let service = super::super::KEYCHAIN_SERVICE;
        let absent = get_generic_password(service, &account).expect_err("fresh unique account");
        assert_eq!(absent.code(), -25300, "other Keychain failures are not absence");
        assert_eq!(create_macos_key(&account, &[11; 32])?, KeyCreation::Created);
        let mut owned = OwnedItem { account, active: true };
        assert_eq!(create_macos_key(&owned.account, &[22; 32])?, KeyCreation::AlreadyExists);
        assert_eq!(get_generic_password(service, &owned.account)?, vec![11; 32]);
        assert_eq!(load_or_create_macos_key(&owned.account)?, [11; 32]);

        // Force the interleaving already covered by the portable race test:
        // this caller observed absence, another caller published, then its
        // actual OS create-only attempt must lose without overwriting.
        let mut reads = 0;
        let resolved = load_or_create(
            || {
                reads += 1;
                if reads == 1 { Ok(None) }
                else { Ok(Some(get_generic_password(service, &owned.account)?)) }
            },
            || Ok([33; 32]),
            |key| create_macos_key(&owned.account, key),
        )?;
        assert_eq!(resolved, [11; 32]);
        assert_eq!(reads, 2);
        assert_eq!(get_generic_password(service, &owned.account)?, vec![11; 32]);
        delete_generic_password(service, &owned.account)?;
        owned.active = false;
        let removed = get_generic_password(service, &owned.account).expect_err("owned item removed");
        assert_eq!(removed.code(), -25300);
        Ok(())
    }
}
